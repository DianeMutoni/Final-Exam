#Summarize the dataset by Continent

continent_summary <- Happiness_corruption %>%
  group_by(continent) %>%
  summarise(Observations = n(),
            Mean_Happiness = mean(happiness_score, na.rm = TRUE),
            Mean_CPI = mean(cpi_score, na.rm = TRUE),
            Mean_GDP = mean(gdp_per_capita, na.rm = TRUE),
            .groups = "drop") %>%
  arrange(desc(Observations))

print(continent_summary, digits = 3,
      caption = "Summary statistics by continent")

#4.4Identify one group of countries with a notably low happiness level.

low_group <-Happiness_corruption %>%
  filter(continent == "Africa", cpi_score < 30)

low_group_countries <- sort(unique(low_group$Country))
low_group_mean <- mean(low_group$happiness_score, na.rm = TRUE)
global_mean <- mean(Happiness_corruption$happiness_score, na.rm = TRUE)
difference <- low_group_mean - global_mean
percent_below <- 100 * (global_mean - low_group_mean) / global_mean
